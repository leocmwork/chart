function renderChart() { const ctx = document.getElementById('bankrollChart').getContext('2d'); let cumulative = 0; const labels = ['Start']; const data = [0]; sesionesData.forEach((s, index) => { cumulative += s.resultado; labels.push(`S${index + 1}`); data.push(cumulative); }); if (chartInstance) chartInstance.destroy(); if (animationTimer) clearInterval(animationTimer); const gradient = ctx.createLinearGradient(0, 0, 0, 400); const isPositive = cumulative >= 0; if (isPositive) { gradient.addColorStop(0, 'rgba(16, 185, 129, 0.4)'); gradient.addColorStop(1, 'rgba(16, 185, 129, 0.0)'); } else { gradient.addColorStop(0, 'rgba(244, 63, 94, 0.4)'); gradient.addColorStop(1, 'rgba(244, 63, 94, 0.0)'); } const mainColor = isPositive ? '#10b981' : '#f43f5e'; chartInstance = new Chart(ctx, { type: 'line', data: { labels: labels, datasets: [{ label: 'Bankroll ($)', data: data, borderColor: mainColor, borderWidth: 3, backgroundColor: gradient, fill: true, tension: 0.35, pointBackgroundColor: mainColor, pointBorderColor: '#0b0f17', pointBorderWidth: 2, pointRadius: 4, pointHoverRadius: 7, pointHoverBackgroundColor: '#fff', pointHoverBorderColor: mainColor, pointHoverBorderWidth: 3 }] }, options: { responsive: true, maintainAspectRatio: false, animation: { x: { type: 'number', easing: 'easeInOutQuart', duration: 2000, from: NaN, // Fuerza el efecto de dibujado desde el eje Y delay(ctx) { if (ctx.type !== 'data' || ctx.xStarted) return 0; ctx.xStarted = true; return ctx.index * 150; // Animación secuencial punto por punto } }, y: { type: 'number', easing: 'easeInOutQuart', duration: 2000 } }, plugins: { legend: { display: false }, tooltip: { backgroundColor: '#1e293b', titleColor: '#94a3b8', bodyColor: '#f8fafc', bodyFont: { weight: 'bold', size: 14 }, borderColor: 'rgba(255,255,255,0.1)', borderWidth: 1, padding: 12, displayColors: false, callbacks: { label: function(context) { return `Bankroll: $${context.parsed.y.toFixed(2)}`; } } } }, scales: { x: { grid: { color: 'rgba(255,255,255,0.03)', drawBorder: false }, ticks: { color: '#64748b', font: { family: 'Inter', size: 11 } } }, y: { grid: { color: 'rgba(255,255,255,0.05)', drawBorder: false }, ticks: { color: '#64748b', font: { family: 'Inter', size: 11 }, callback: function(value) { return '$' + value; } } } } } }); // Reinicia el estado visual y vuelve a animar cada 8 segundos animationTimer = setInterval(() => { if (chartInstance && !isHovered) { chartInstance.reset(); // Resetea la posición inicial chartInstance.update(); // Ejecuta el dibujado progresivo } }, 8000); }
