// Inicializa el gráfico base con degradado de 0.4 (arriba) a 0.9 (abajo) function initChart() { const ctx = document.getElementById('bankrollChart').getContext('2d'); const { cumulative } = getChartFullData(); if (chartInstance) chartInstance.destroy(); if (loopIntervalId) clearInterval(loopIntervalId); const isPositive = cumulative >= 0; const gradient = ctx.createLinearGradient(0, 0, 0, 400); if (isPositive) { // Verde: 0.4 de opacidad arriba -> 0.9 de opacidad abajo gradient.addColorStop(0, 'rgba(16, 185, 129, 0.4)'); gradient.addColorStop(1, 'rgba(16, 185, 129, 0.9)'); } else { // Rojo: 0.4 de opacidad arriba -> 0.9 de opacidad abajo gradient.addColorStop(0, 'rgba(244, 63, 94, 0.4)'); gradient.addColorStop(1, 'rgba(244, 63, 94, 0.9)'); } const mainColor = isPositive ? '#10b981' : '#f43f5e'; chartInstance = new Chart(ctx, { type: 'line', data: { labels: [], datasets: [{ label: 'Bankroll ($)', data: [], borderColor: mainColor, borderWidth: 3, backgroundColor: gradient, fill: true, tension: 0.35, pointBackgroundColor: mainColor, pointBorderColor: '#0b0f17', pointBorderWidth: 2, pointRadius: 4, pointHoverRadius: 7, pointHoverBackgroundColor: '#fff', pointHoverBorderColor: mainColor, pointHoverBorderWidth: 3 }] }, options: { responsive: true, maintainAspectRatio: false, animation: { duration: 300 }, plugins: { legend: { display: false }, tooltip: { backgroundColor: '#1e293b', titleColor: '#94a3b8', bodyColor: '#f8fafc', bodyFont: { weight: 'bold', size: 14 }, borderColor: 'rgba(255,255,255,0.1)', borderWidth: 1, padding: 12, displayColors: false, callbacks: { label: function(context) { return `Bankroll: $${context.parsed.y.toFixed(2)}`; } } } }, scales: { x: { grid: { color: 'rgba(255,255,255,0.03)', drawBorder: false }, ticks: { color: '#64748b', font: { family: 'Inter', size: 11 } } }, y: { grid: { color: 'rgba(255,255,255,0.05)', drawBorder: false }, ticks: { color: '#64748b', font: { family: 'Inter', size: 11 }, callback: function(value) { return '$' + value; } } } } } }); startProgressiveLoop(); }
